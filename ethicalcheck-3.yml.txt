name: API-Security-Scan
on:
  push:
    branches: ["main"]
  pull_request:
    branches: ["main"]
  schedule:
    - cron: '38 20 * * 3'
  workflow_dispatch:

# Minimum permissions — principle of least privilege
permissions:
  contents: read

jobs:
  zap_scan:
    permissions:
      security-events: write  # Required to upload SARIF to GitHub Security tab
      actions: read           # Required to read workflow context
    runs-on: ubuntu-latest
    name: Scan the API
    steps:
      - name: Checkout
        uses: actions/checkout@11bd71901bbe5b1630ceea73d27597364c9af683 # v4

      - name: ZAP API Scan
        uses: zaproxy/action-api-scan@v0.7.0
        with:
          # No token — OWASP ZAP scans the public target URL directly.
          # Zero secret references. Scan results output to Actions console only.
          format: openapi
          target: 'http://netbanking.apisec.ai:8080/v2/api-docs'
          rules_file_name: '.zap/rules.tsv'
          # cmd_options '-a' removed — Ajax spider not needed for OpenAPI scans
